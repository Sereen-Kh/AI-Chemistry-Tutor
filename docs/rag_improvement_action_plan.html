🔍 Root cause — one number is breaking everything
Your hybrid scoring uses 65% lexical weight + 35% vector weight. Arabic morphology makes keyword matching unreliable — "الأملاح" doesn't match "أملاح". The lexical score collapses to near zero, dragging the combined score down to 28-44% even when the vector similarity is actually 72-80%. Flip those weights and you fix the core problem instantly.

# Add these two functions to rag.py DEFINITION_TRIGGERS = ["ما هو", "ما هي", "عرّف", "اشرح", "فسّر", "ما المقصود", "ما الفرق"] CALCULATION_TRIGGERS = ["احسب", "أوجد", "حل", "طبّق", "وازن"] def detect_intent(query: str) -> str: """Classify Arabic query as definition, calculation, or default.""" if any(t in query for t in DEFINITION_TRIGGERS): return "definition" if any(t in query for t in CALCULATION_TRIGGERS): return "calculation" return "default" def apply_intent_boost( score: float, content_type: str, intent: str ) -> float: """Boost or penalise a chunk score based on its type vs query intent.""" if intent == "definition": if content_type in ("definition", "text"): return score + 0.08 # prefer definition chunks if content_type == "exercise": return score - 0.12 # strongly penalise exercises elif intent == "calculation": if content_type in ("exercise", "result"): return score + 0.08 # exercises are good for calculations return score # In your main retrieval function, use it like this: def retrieve_chunks(query: str, ...) -> list: intent = detect_intent(query) # ... your existing vector search ... # After getting candidates, apply boost + filter: boosted = [] for chunk in candidates: final_score = hybrid_score(chunk.vector_sim, chunk.lexical_sim) final_score = apply_intent_boost( final_score, chunk.content_type, intent ) if final_score >= MIN_SIMILARITY: chunk.similarity_score = final_score boosted.append(chunk) return sorted(boosted, key=lambda c: c.similarity_score, reverse=True)[:6]
